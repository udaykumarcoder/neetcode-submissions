2. Exact Python code — highlighted line

1 LIS = [1] * len(nums) 2 3 for i in range(len(nums) - 1, -1, -1): 4 for j in range(i + 1, len(nums)): 5 if nums[i] < nums[j]: 6 LIS[i] = max(LIS[i], 1 + LIS[j]) 7 8 return max(LIS)
