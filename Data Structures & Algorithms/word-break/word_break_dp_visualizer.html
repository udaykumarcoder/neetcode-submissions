2. Exact Python code — highlighted line

1 dp = [False] * (len(s) + 1) 2 dp[len(s)] = True 3 4 for i in range(len(s) - 1, -1, -1): 5 for w in wordDict: 6 if (i + len(w) <= len(s) and 7 s[i : i + len(w)] == w): 8 dp[i] = dp[i + len(w)] 9 if dp[i]: 10 break 11 return dp[0]
